from typing import List


def fibonacci_sequence(n: int) -> List[int]:
    """
    计算斐波那契数列的前 N 项。

    斐波那契数列定义：F(0) = 0, F(1) = 1, F(n) = F(n-1) + F(n-2)（n >= 2）。
    本函数返回从 F(0) 开始的前 n 项列表。

    参数:
        n (int): 需要生成的项数，必须为正整数。

    返回:
        List[int]: 包含前 n 个斐波那契数的列表。

    异常:
        ValueError: 当 n <= 0 时抛出，提示参数必须为正整数。

    示例:
        >>> fibonacci_sequence(5)
        [0, 1, 1, 2, 3]
        >>> fibonacci_sequence(1)
        [0]
    """
    if n <= 0:
        raise ValueError("参数 n 必须为正整数，当前收到 n = {}".format(n))

    # 初始化前两项
    fib_list: List[int] = [0, 1]

    # 当 n == 1 时，只返回 [0]
    if n == 1:
        return fib_list[:1]

    # 从第 3 项开始生成（索引 2 对应 F(2)）
    for i in range(2, n):
        next_fib = fib_list[i - 1] + fib_list[i - 2]
        fib_list.append(next_fib)

    return fib_list
