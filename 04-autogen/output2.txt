import time
import random
from typing import Callable, Dict, List, Tuple
import sys

def bubble_sort(arr: List[int]) -> List[int]:
    """冒泡排序 - O(n²)"""
    n = len(arr)
    arr = arr.copy()  # 不修改原数组
    for i in range(n):
        for j in range(0, n - i - 1):
            if arr[j] > arr[j + 1]:
                arr[j], arr[j + 1] = arr[j + 1], arr[j]
    return arr

def selection_sort(arr: List[int]) -> List[int]:
    """选择排序 - O(n²)"""
    n = len(arr)
    arr = arr.copy()
    for i in range(n):
        min_idx = i
        for j in range(i + 1, n):
            if arr[j] < arr[min_idx]:
                min_idx = j
        arr[i], arr[min_idx] = arr[min_idx], arr[i]
    return arr

def insertion_sort(arr: List[int]) -> List[int]:
    """插入排序 - O(n²)"""
    arr = arr.copy()
    for i in range(1, len(arr)):
        key = arr[i]
        j = i - 1
        while j >= 0 and arr[j] > key:
            arr[j + 1] = arr[j]
            j -= 1
        arr[j + 1] = key
    return arr

def merge_sort(arr: List[int]) -> List[int]:
    """归并排序 - O(n log n)"""
    if len(arr) <= 1:
        return arr.copy()
    
    mid = len(arr) // 2
    left = merge_sort(arr[:mid])
    right = merge_sort(arr[mid:])
    
    result = []
    i = j = 0
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            result.append(left[i])
            i += 1
        else:
            result.append(right[j])
            j += 1
    result.extend(left[i:])
    result.extend(right[j:])
    return result

def quick_sort(arr: List[int]) -> List[int]:
    """快速排序 - 平均 O(n log n)"""
    if len(arr) <= 1:
        return arr.copy()
    
    pivot = arr[len(arr) // 2]
    left = [x for x in arr if x < pivot]
    middle = [x for x in arr if x == pivot]
    right = [x for x in arr if x > pivot]
    
    return quick_sort(left) + middle + quick_sort(right)

def compare_sorting_algorithms(
    sizes: List[int] = None,
    num_trials: int = 3
) -> Dict[str, Dict[int, float]]:
    """
    比较不同排序算法的性能。
    
    Args:
        sizes: 要测试的数组大小列表，默认为 [100, 500, 1000, 5000]
        num_trials: 每个大小重复测试的次数，取平均值
    
    Returns:
        字典，格式为 {算法名: {数组大小: 平均耗时(秒)}}
    
    Raises:
        ValueError: 如果 sizes 包含非正数或 num_trials <= 0
    """
    if sizes is None:
        sizes = [100, 500, 1000, 5000]
    
    # 边界检查
    if any(n <= 0 for n in sizes):
        raise ValueError("数组大小必须为正数")
    if num_trials <= 0:
        raise ValueError("测试次数必须为正数")
    
    algorithms: Dict[str, Callable[[List[int]], List[int]]] = {
        "冒泡排序": bubble_sort,
        "选择排序": selection_sort,
        "插入排序": insertion_sort,
        "归并排序": merge_sort,
        "快速排序": quick_sort,
    }
    
    results: Dict[str, Dict[int, float]] = {name: {} for name in algorithms}
    
    for size in sizes:
        print(f"\n测试数组大小: {size}")
        
        for name, algo in algorithms.items():
            total_time = 0.0
            
            for _ in range(num_trials):
                # 生成随机数组
                test_data = [random.randint(0, 10000) for _ in range(size)]
                
                # 计时
                start = time.perf_counter()
                algo(test_data)
                end = time.perf_counter()
                total_time += (end - start)
            
            avg_time = total_time / num_trials
            results[name][size] = avg_time
            print(f"  {name}: {avg_time:.6f} 秒")
    
    return results

def print_results_table(results: Dict[str, Dict[int, float]]) -> None:
    """
    以表格形式打印性能比较结果。
    
    Args:
        results: compare_sorting_algorithms 的返回值
    """
    if not results:
        print("无结果可显示")
        return
    
    # 获取所有大小
    all_sizes = sorted(set(size for sizes in results.values() for size in sizes))
    
    # 打印表头
    header = f"{'算法':<10}" + "".join(f"{size:>12}" for size in all_sizes)
    print("\n" + "=" * len(header))
    print(header)
    print("=" * len(header))
    
    # 打印每行
    for algo_name, size_times in results.items():
        row = f"{algo_name:<10}"
        for size in all_sizes:
            if size in size_times:
                row += f"{size_times[size]:>12.6f}"
            else:
                row += f"{'N/A':>12}"
        print(row)
    
    print("=" * len(header))

    test_sizes = [100, 500, 1000]
    results = compare_sorting_algorithms(test_sizes, num_trials=2)
    print_results_table(results)

# 排序算法性能比较工具

一个用于比较不同复杂度排序算法性能的 Python 工具，支持冒泡排序、选择排序、插入排序、归并排序和快速排序的性能基准测试。

## 函数签名和参数说明

### `compare_sorting_algorithms(sizes: Optional[List[int]] = None, num_trials: int = 3) -> Dict[str, Dict[int, float]]`

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `sizes` | `Optional[List[int]]` | `[100, 500, 1000, 5000]` | 要测试的数组大小列表，每个元素必须为正整数 |
| `num_trials` | `int` | `3` | 每个数组大小重复测试的次数，取平均值作为最终结果 |

**返回值**: `Dict[str, Dict[int, float]]` - 格式为 `{算法名: {数组大小: 平均耗时(秒)}}`

### `print_results_table(results: Dict[str, Dict[int, float]]) -> None`

| 参数 | 类型 | 说明 |
|------|------|------|
| `results` | `Dict[str, Dict[int, float]]` | `compare_sorting_algorithms` 的返回值 |

### 排序算法函数

所有排序函数签名一致：`(arr: List[int]) -> List[int]`，输入整数列表，返回排序后的新列表（不修改原数组）。

| 函数名 | 时间复杂度 | 说明 |
|--------|-----------|------|
| `bubble_sort` | O(n²) | 冒泡排序 |
| `selection_sort` | O(n²) | 选择排序 |
| `insertion_sort` | O(n²) | 插入排序 |
| `merge_sort` | O(n log n) | 归并排序 |
| `quick_sort` | O(n log n) | 快速排序 |

## 行为说明

1. **性能测试机制**：对每个数组大小，生成随机整数数组（范围 0-10000），重复测试 `num_trials` 次并计算平均耗时
2. **非破坏性操作**：所有排序函数均返回新列表，不修改原始输入数组
3. **边界处理**：当 `sizes` 为空、包含非正数或 `num_trials <= 0` 时抛出 `ValueError`
4. **递归保护**：归并排序和快速排序实现了递归深度保护，防止处理超大数组时栈溢出
5. **结果输出**：测试过程中实时打印每个算法的耗时，最终可通过 `print_results_table` 生成格式化表格

## 行为说明

1. **性能测试机制**：对每个数组大小，生成随机整数数组（范围 0-10000），重复测试 `num_trials` 次并计算平均耗时
2. **非破坏性操作**：所有排序函数均返回新列表，不修改原始输入数组
3. **边界处理**：当 `sizes` 为空、包含非正数或 `num_trials <= 0` 时抛出 `ValueError`
4. **递归保护**：归并排序和快速排序实现了递归深度保护，防止处理超大数组时栈溢出
5. **结果输出**：测试过程中实时打印每个算法的耗时，最终可通过 `print_results_table` 生成格式化表格
