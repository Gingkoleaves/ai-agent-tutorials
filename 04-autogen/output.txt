import random
import time
from typing import List


def compare_sorting_algorithms(n: int) -> None:
    """
    比较不同复杂度排序算法的性能。

    生成包含 n 个随机整数的列表，分别使用冒泡排序（O(n^2)）、
    归并排序（O(n log n)）、快速排序（O(n log n)）和 Python 内置
    sorted（Timsort，O(n log n)）对相同数据进行排序，测量并打印
    各算法的运行时间。

    参数:
        n: 要排序的随机整数个数。必须为正整数。

    返回:
        None。函数直接打印性能比较结果。

    说明:
        如果 n <= 0，打印错误信息并直接返回。
    """
    if n <= 0:
        print("错误：n 必须是正整数。")
        return

    # 生成原始随机列表
    base_list = [random.randint(0, 100000) for _ in range(n)]

    # 冒泡排序（返回新列表）
    def bubble_sort(arr: List[int]) -> List[int]:
        """冒泡排序，返回排序后的新列表。"""
        arr = arr[:]
        length = len(arr)
        for i in range(length):
            swapped = False
            for j in range(0, length - i - 1):
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
                    swapped = True
            if not swapped:
                break
        return arr

    # 归并排序（返回新列表）
    def merge_sort(arr: List[int]) -> List[int]:
        """归并排序，返回排序后的新列表。"""

        def merge(left: List[int], right: List[int]) -> List[int]:
            result = []
            i = j = 0
            while i < len(left) and j < len(right):
                if left[i] <= right[j]:
                    result.append(left[i])
                    i += 1
                else:
                    result.append(right[j])
                    j += 1
            result.extend(left[i:])
            result.extend(right[j:])
            return result

        if len(arr) <= 1:
            return arr[:]
        mid = len(arr) // 2
        left = merge_sort(arr[:mid])
        right = merge_sort(arr[mid:])
        return merge(left, right)

    # 快速排序（返回新列表）
    def quick_sort(arr: List[int]) -> List[int]:
        """快速排序，返回排序后的新列表。"""
        if len(arr) <= 1:
            return arr[:]
        pivot = arr[len(arr) // 2]
        left = [x for x in arr if x < pivot]
        middle = [x for x in arr if x == pivot]
        right = [x for x in arr if x > pivot]
        return quick_sort(left) + middle + quick_sort(right)

    # 参与比较的算法及其复杂度
    algorithms = [
        ("Bubble Sort", "O(n^2)", bubble_sort),
        ("Merge Sort", "O(n log n)", merge_sort),
        ("Quick Sort", "O(n log n)", quick_sort),
        ("Timsort (built-in)", "O(n log n)", sorted),
    ]

    expected = sorted(base_list)  # 用于验证正确性

    print(f"比较排序算法性能，数据规模 n = {n}")
    print("-" * 70)
    print(f"{'算法':<25} {'时间复杂度':<18} {'耗时(秒)':>12}")
    print("-" * 70)

    for name, complexity, func in algorithms:
        start = time.perf_counter()
        result = func(base_list)
        elapsed = time.perf_counter() - start

        # 验证排序结果正确性
        if result != expected:
            print(f"{name:<25} {complexity:<18} {elapsed:>12.6f}  [错误]")
        else:
            print(f"{name:<25} {complexity:<18} {elapsed:>12.6f}")

    print("-" * 70)


if __name__ == "__main__":
    compare_sorting_algorithms(1000)

# 排序算法性能比较工具

## 概述

`compare_sorting_algorithms` 是一个 Python 函数，用于比较不同时间复杂度排序算法在相同随机数据集上的实际运行性能。它分别使用**冒泡排序**（O(n²)）、**归并排序**（O(n log n)）、**快速排序**（O(n log n)）和 **Python 内置 `sorted`**（Timsort，O(n log n)）对同一列表进行排序，输出每个算法的耗时，并校验排序结果的正确性。

## 函数签名

```python
def compare_sorting_algorithms(n: int) -> None
```

## 参数说明

| 参数 | 类型 | 说明 |
|------|------|------|
| `n`  | `int` | 要排序的随机整数个数。必须为正整数。 |

## 返回值

无返回值（`None`）。函数直接将性能比较结果打印到控制台。

## 行为说明

1. 如果 `n <= 0`，打印错误信息 `"错误：n 必须是正整数。"` 并立即返回。
2. 生成 `n` 个范围在 `[0, 100000]` 的随机整数作为原始列表。
3. 对同一份原始列表分别执行四种排序算法，并记录运行时间（使用高精度计时器 `time.perf_counter`）。
4. 使用 Python 内置 `sorted` 的结果作为基准，验证其他算法的排序结果是否正确；若结果不一致，会在输出中标记 `[错误]`。
5. 打印包含算法名称、时间复杂度和运行时间的表格。

## 输出示例

```
比较排序算法性能，数据规模 n = 1000

算法                       时间复杂度          耗时(秒)

Bubble Sort                O(n^2)                0.123456
Merge Sort                 O(n log n)            0.001234
Quick Sort                 O(n log n)            0.000987
Timsort (built-in)         O(n log n)            0.000654
----------------------------------------------------------------------
```

> 实际耗时取决于机器性能和数据规模。

## 注意事项

- 冒泡排序在 `n` 较大时非常耗时，建议仅在小规模数据（如 `n <= 2000`）下运行。
- 所有排序算法均返回新列表，不会修改传入的原始数据。
- 内置 `sorted` 函数本身就用于生成期望结果，因此其正确性验证始终通过。
